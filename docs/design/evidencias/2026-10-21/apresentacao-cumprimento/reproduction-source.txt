import assert from 'node:assert/strict';
import {readFileSync,writeFileSync} from 'node:fs';
import {join,resolve} from 'node:path';
import {createHash} from 'node:crypto';
import {summarizePerformance} from '../../../app/test/support/performance-samples.ts';
import {attributeToggleLatency} from '../../../app/test/support/toggle-latency.ts';
import {validateObservationMode} from '../../../app/test/support/performance-mode.ts';
const folder=resolve(import.meta.dirname,'../../../docs/design/evidencias/2026-10-21/apresentacao-cumprimento/diagnostico2');
const read=(name:string):unknown=>JSON.parse(readFileSync(join(folder,name),'utf8'));
const frames=read('native-frames.json'),marks=read('javascript-milestones.json'),chunks=read('javascript-profiles.json');
assert.ok(Array.isArray(frames)&&Array.isArray(marks)&&Array.isArray(chunks));
assert.deepEqual(summarizePerformance(frames),read('performance-summary.json'));
assert.deepEqual(attributeToggleLatency(frames,marks),read('toggle-attribution.json'));
assert.deepEqual(validateObservationMode('lightweight',frames,marks,chunks),[]);
assert.deepEqual(read('observation-mode.json'),{mode:'lightweight',probe:4,synchronousClock:false,schedulerTrace:true,presentationTrace:true,validated:true});
const result=read('remote-result.json');assert.deepEqual(result,{habitCount:1,done:true,totalXp:10,coldStarts:10,timedVisualChanges:20,persistedAfterRestart:true});
const files=['native-frames.json','javascript-milestones.json','javascript-profiles.json','performance-summary.json','toggle-attribution.json','observation-mode.json','remote-result.json'];
writeFileSync(join(folder,'reproduction.json'),JSON.stringify({passed:true,sameGestureAndFrameValidated:true,modeAndIndependentPeerValidated:true,inputsSha256:Object.fromEntries(files.map(name=>[name,createHash('sha256').update(readFileSync(join(folder,name))).digest('hex')]))},null,2)+'\n');
console.log('Replayed native frame/PressEvent attribution, explicit mode and independent peer: passed.');
